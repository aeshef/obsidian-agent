<%*
try {
  const title = await tp.system.prompt("Название рецепта");
  if (!title) { await tp.app.vault.delete(tp.file.find_tfile(tp.file.title)); return; }

  const cuisine = await tp.system.prompt("Кухня (англ., например: italian)", "");
  const kind = await tp.system.prompt("Тип блюда (англ., например: appetizer)", "");
  const time = await tp.system.prompt("Время приготовления (напр. 30m)", "");
  const link = await tp.system.prompt("Ссылка (опционально)", "");
  const ingredientsRaw = await tp.system.prompt("Ингредиенты (каждый с новой строки)", "");
  const ingredients = (ingredientsRaw || '').split(/\n+/).map(s => s.trim()).filter(Boolean);

  const slug = (s) => s.replace(/[^\p{L}\p{N}\s-_]/gu, "").trim().replace(/\s+/g, "_");
  const baseDir = `700_База_Данных/Рецепты`;
  const baseName = slug(title);
  let filePath = `${baseDir}/${baseName}.md`;
  if (tp.app.vault.getAbstractFileByPath(filePath)) {
    filePath = `${baseDir}/${baseName}-i.md`;
  }

  const tags = [];
  if (cuisine) tags.push(`cuisine/${cuisine.toLowerCase()}`);
  if (kind) tags.push(`kind/${kind.toLowerCase()}`);
  tags.push('domain/food');

  const created = tp.date.now("YYYY-MM-DD");
  const fm = [
    '---',
    `type: "рецепт"`,
    `title: "${title}"`,
    `created: ${created}`,
    'tags:',
    ...(tags.length ? tags.map(t => `  - ${t}`) : ['  -']),
    'attachments:',
    '  links:',
    link ? `  - "${link}"` : '  -',
    '  files:',
    'source: ""',
    'form: "text"',
    `cuisine: "${cuisine}"`,
    `kind: "${kind}"`,
    `time: "${time}"`,
    'ingredients:',
    ...(ingredients.length ? ingredients.map(i => `  - ${i}`) : ['  -']),
    'raw_dir: ""',
    '---'
  ].join('\n');

  const body = [
    `# ${title}`,
    '',
    '## Ингредиенты',
    ...(ingredients.length ? ingredients.map(i => `- ${i}`) : ['- ']),
    '',
    '## Шаги',
    '1. ',
    '',
    '## Ссылки',
    link ? `- ${link}` : '- ',
    '',
    '## Файлы'
  ].join('\n');

  const content = `${fm}\n\n${body}\n`;
  const newFile = await tp.app.vault.create(filePath, content);
  const currentFile = tp.file.find_tfile(tp.file.title); if (currentFile) await tp.app.vault.delete(currentFile);
  await tp.app.workspace.getLeaf().openFile(newFile);
  new Notice(`✅ Рецепт "${title}" добавлен!`);
} catch (error) {
  new Notice(`❌ Ошибка: ${error.message}`);
}
%>
