<%*
try {
  const title = await tp.system.prompt("Краткое имя/тема цитаты");
  if (!title) { await tp.app.vault.delete(tp.file.find_tfile(tp.file.title)); return; }

  const text = await tp.system.prompt("Текст цитаты", "");
  const author = await tp.system.prompt("Автор (опц.)", "");
  const nsfw = await tp.system.suggester(["false", "true"],["false","true"]) || "false";
  const topic = await tp.system.prompt("Тема (topic/...) — опц.", "");

  const slug = (s) => s.replace(/[^\p{L}\p{N}\s-_]/gu, "").trim().replace(/\s+/g, "_");
  const baseDir = `700_База_Данных/Цитаты`;
  const baseName = slug(title);
  let filePath = `${baseDir}/${baseName}.md`;
  while (tp.app.vault.getAbstractFileByPath(filePath)) { filePath = `${baseDir}/${baseName}-i.md`; }

  const tags = [];
  if (topic) tags.push(`topic/${topic.toLowerCase()}`);

  const created = tp.date.now("YYYY-MM-DD");
  const fm = [
    '---',
    `type: "цитата"`,
    `title: "${title}"`,
    `created: ${created}`,
    'tags:',
    ...tags.map(t => `  - ${t}`),
    'attachments:',
    '  links:',
    '  files:',
    'form: "text"',
    `text: "${text}"`,
    `author: "${author}"`,
    `nsfw: ${nsfw}`,
    'raw_dir: ""',
    '---'
  ].join('\n');

  const body = [
    `# ${title}`,
    '',
    `> ${text}`,
    '',
    `- Автор: ${author}`,
    '- Исходные файлы: ',
  ].join('\n');

  const content = `${fm}\n\n${body}\n`;
  const newFile = await tp.app.vault.create(filePath, content);
  const currentFile = tp.file.find_tfile(tp.file.title); if (currentFile) await tp.app.vault.delete(currentFile);
  await tp.app.workspace.getLeaf().openFile(newFile);
  new Notice(`✅ Цитата "${title}" добавлена!`);
} catch (error) { new Notice(`❌ Ошибка: ${error.message}`); }
%>
