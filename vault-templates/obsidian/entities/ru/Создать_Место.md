<%*
try {
  const title = await tp.system.prompt("Название места");
  if (!title) { await tp.app.vault.delete(tp.file.find_tfile(tp.file.title)); return; }

  const city = await tp.system.prompt("Город", "");
  const address = await tp.system.prompt("Адрес (опционально)", "");
  const category = await tp.system.prompt("Категория (category/...)", "");
  const link = await tp.system.prompt("Ссылка (карта)", "");
  const latitude = await tp.system.prompt("Широта (опц.)", "");
  const longitude = await tp.system.prompt("Долгота (опц.)", "");
  const event_date = await tp.system.prompt("Дата события YYYY-MM-DD (опц.)", "");
  const event_time = await tp.system.prompt("Время события HH:MM (опц.)", "");

  const slug = (s) => s.replace(/[^\p{L}\p{N}\s-_]/gu, "").trim().replace(/\s+/g, "_");
  const baseDir = `700_База_Данных/Места`;
  const baseName = slug(title);
  let filePath = `${baseDir}/${baseName}.md`;
  while (tp.app.vault.getAbstractFileByPath(filePath)) { filePath = `${baseDir}/${baseName}-i.md`; }

  const tags = [];
  if (category) tags.push(`category/${category}`);
  if (city) tags.push(`city/${city}`);

  const created = tp.date.now("YYYY-MM-DD");
  const fm = [
    '---',
    `type: "место"`,
    `title: "${title}"`,
    `created: ${created}`,
    'tags:',
    ...tags.map(t => `  - ${t}`),
    'attachments:',
    '  links:',
    link ? `  - "${link}"` : '  -',
    '  files:',
    'form: "link"',
    `latitude: ${latitude || ''}`,
    `longitude: ${longitude || ''}`,
    `address: "${address}"`,
    `city: "${city}"`,
    `category: "${category}"`,
    `event_date: "${event_date}"`,
    `event_time: "${event_time}"`,
    'raw_dir: ""',
    '---'
  ].join('\n');

  const body = [
    `# ${title}`,
    '',
    '- Исходные файлы: ',
    '',
    '## Заметки о месте',
    '- '
  ].join('\n');

  const content = `${fm}\n\n${body}\n`;
  const newFile = await tp.app.vault.create(filePath, content);
  const currentFile = tp.file.find_tfile(tp.file.title); if (currentFile) await tp.app.vault.delete(currentFile);
  await tp.app.workspace.getLeaf().openFile(newFile);
  new Notice(`✅ Место "${title}" добавлено!`);
} catch (error) {
  new Notice(`❌ Ошибка: ${error.message}`);
}
%>
