try {
  const title = await tp.system.prompt("Название песни");
  if (!title) { await tp.app.vault.delete(tp.file.find_tfile(tp.file.title)); return; }

  const author = await tp.system.prompt("Исполнитель (опц.)", "");
  const status = await tp.system.suggester(["planned", "in-progress", "done"],["planned", "in-progress", "done"]) || "planned";
  const source = await tp.system.prompt("Источник (spotify/youtube/...) (опц.)", "");
  const link = await tp.system.prompt("Ссылка (опц.)", "");
  const topic = await tp.system.prompt("Тема (topic/...) — опц.", "music");

  const slug = (s) => s.replace(/[^\p{L}\p{N}\s-_]/gu, "").trim().replace(/\s+/g, "_");
  const baseDir = `700_База_Данных/Песни`;
  const baseName = slug(title);
  let filePath = `${baseDir}/${baseName}.md`;
  while (tp.app.vault.getAbstractFileByPath(filePath)) { filePath = `${baseDir}/${baseName}-i.md`; }

  const tags = [];
  if (topic) tags.push(`topic/${topic.toLowerCase()}`);
  tags.push(`status/${status}`);
  if (source) tags.push(`source/${source.toLowerCase().replace(/\s+/g, '_')}`);

  const created = tp.date.now("YYYY-MM-DD");
  const fm = [
    '---',
    `type: "песня"`,
    `title: "${title}"`,
    `created: ${created}`,
    'tags:',
    ...tags.map(t => `  - ${t}`),
    'attachments:',
    '  links:',
    link ? `  - "${link}"` : '  -',
    '  files:',
    `source: "${source}"`,
    'form: "link"',
    `status: "${status}"`,
    `author: "${author}"`,
    'rating: ',
    'raw_dir: ""',
    '---'
  ].join('\n');

  const body = [
    `# ${title}`,
    '',
    '- Исходные файлы: ',
    '',
    '## Заметки',
    '- '
  ].join('\n');

  const content = `${fm}\n\n${body}\n`;
  const newFile = await tp.app.vault.create(filePath, content);
  const currentFile = tp.file.find_tfile(tp.file.title); if (currentFile) await tp.app.vault.delete(currentFile);
  await tp.app.workspace.getLeaf().openFile(newFile);
  new Notice(`✅ Песня "${title}" добавлена!`);
} catch (error) {
  new Notice(`❌ Ошибка: ${error.message}`);
}
