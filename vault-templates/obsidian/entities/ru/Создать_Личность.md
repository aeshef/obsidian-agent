<%*
try {
  const title = await tp.system.prompt("Название вайб‑хаба/личности");
  if (!title) { await tp.app.vault.delete(tp.file.find_tfile(tp.file.title)); return; }

  const traitsRaw = await tp.system.prompt("Черты (каждая с новой строки) (опц.)", "");
  const inspRaw = await tp.system.prompt("Инспирации (каждая с новой строки) (опц.)", "");
  const linksRaw = await tp.system.prompt("Ссылки (каждая с новой строки) (опц.)", "");

  const traits = (traitsRaw || '').split(/\n+/).map(s => s.trim()).filter(Boolean);
  const inspirations = (inspRaw || '').split(/\n+/).map(s => s.trim()).filter(Boolean);
  const links = (linksRaw || '').split(/\n+/).map(s => s.trim()).filter(Boolean);

  const slug = (s) => s.replace(/[^\p{L}\p{N}\s-_]/gu, "").trim().replace(/\s+/g, "_");
  const baseDir = `700_База_Данных/Личности`;
  const baseName = slug(title);
  let filePath = `${baseDir}/${baseName}.md`;
  while (tp.app.vault.getAbstractFileByPath(filePath)) { filePath = `${baseDir}/${baseName}-i.md`; }

  const tags = [];

  const created = tp.date.now("YYYY-MM-DD");
  const fm = [
    '---',
    `type: "личность"`,
    `title: "${title}"`,
    `created: ${created}`,
    'tags:',
    ...tags.map(t => `  - ${t}`),
    'attachments:',
    '  links:',
    ...links.map(l => `  - "${l}"`),
    '  files:',
    'traits:',
    ...traits.map(tr => `  - ${tr}`),
    'inspirations:',
    ...inspirations.map(insp => `  - ${insp}`),
    'raw_dir: ""',
    '---'
  ].join('\n');

  const body = [
    `# ${title}`,
    '',
    '- Исходные файлы: ',
    '',
    '## Мудборд',
    '- '
  ].join('\n');

  const content = `${fm}\n\n${body}\n`;
  const newFile = await tp.app.vault.create(filePath, content);
  const currentFile = tp.file.find_tfile(tp.file.title); if (currentFile) await tp.app.vault.delete(currentFile);
  await tp.app.workspace.getLeaf().openFile(newFile);
  new Notice(`✅ Личность/вайб "${title}" добавлена!`);
} catch (error) { new Notice(`❌ Ошибка: ${error.message}`); }
%>
