<%*
try {
  const title = await tp.system.prompt("Название заметки");
  if (!title) { await tp.app.vault.delete(tp.file.find_tfile(tp.file.title)); return; }

  const summary = await tp.system.prompt("Краткое резюме (опц.)", "");
  const insightsRaw = await tp.system.prompt("Инсайты (каждый с новой строки) (опц.)", "");
  const linksRaw = await tp.system.prompt("Ссылки (каждая с новой строки) (опц.)", "");
  const topic = await tp.system.prompt("Тема (topic/...) — опц.", "");
  const domain = await tp.system.prompt("Домен (domain/...) — опц.", "");

  const insights = (insightsRaw || '').split(/\n+/).map(s => s.trim()).filter(Boolean);
  const links = (linksRaw || '').split(/\n+/).map(s => s.trim()).filter(Boolean);

  const slug = (s) => s.replace(/[^\p{L}\p{N}\s-_]/gu, "").trim().replace(/\s+/g, "_");
  const baseDir = `700_База_Данных/Заметки`;
  const baseName = slug(title);
  let filePath = `${baseDir}/${baseName}.md`;
  while (tp.app.vault.getAbstractFileByPath(filePath)) { filePath = `${baseDir}/${baseName}-i.md`; }

  const tags = [];
  if (domain) tags.push(`domain/${domain.toLowerCase()}`);
  if (topic) tags.push(`topic/${topic.toLowerCase()}`);

  const created = tp.date.now("YYYY-MM-DD");
  const fm = [
    '---',
    `type: "заметка"`,
    `title: "${title}"`,
    `created: ${created}`,
    'tags:',
    ...tags.map(t => `  - ${t}`),
    'attachments:',
    '  links:',
    ...links.map(l => `  - "${l}"`),
    '  files:',
    'source: ""',
    'form: "text"',
    `summary: "${summary}"`,
    'insights:',
    ...insights.map(i => `  - ${i}`),
    'raw_dir: ""',
    '---'
  ].join('\n');

  const body = [
    `# ${title}`,
    '',
    '- Исходные файлы: ',
    '',
    '## Краткое резюме',
    `- ${summary}`,
    '',
    '## Инсайты',
    '- '
  ].join('\n');

  const content = `${fm}\n\n${body}\n`;
  const newFile = await tp.app.vault.create(filePath, content);
  const currentFile = tp.file.find_tfile(tp.file.title); if (currentFile) await tp.app.vault.delete(currentFile);
  await tp.app.workspace.getLeaf().openFile(newFile);
  new Notice(`✅ Заметка "${title}" добавлена!`);
} catch (error) { new Notice(`❌ Ошибка: ${error.message}`); }
%>
