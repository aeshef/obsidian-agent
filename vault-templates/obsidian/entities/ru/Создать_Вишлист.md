<%*
try {
  const title = await tp.system.prompt("Название товара/желания");
  if (!title) { await tp.app.vault.delete(tp.file.find_tfile(tp.file.title)); return; }

  const priority = await tp.system.suggester(["high", "medium", "low"],["high", "medium", "low"]) || "medium";
  const price = await tp.system.prompt("Цена (опц.)", "");
  const reason = await tp.system.prompt("Почему хочу (опц.)", "");
  const link = await tp.system.prompt("Ссылка на товар (опц.)", "");
  const category = await tp.system.prompt("Категория (category/...)", "");
  const source = await tp.system.prompt("Источник (ozon/wb/...) (опц.)", "");

  const slug = (s) => s.replace(/[^\p{L}\p{N}\s-_]/gu, "").trim().replace(/\s+/g, "_");
  const baseDir = `700_База_Данных/Вишлист`;
  const baseName = slug(title);
  let filePath = `${baseDir}/${baseName}.md`;
  while (tp.app.vault.getAbstractFileByPath(filePath)) { filePath = `${baseDir}/${baseName}-i.md`; }

  const tags = [`priority/${priority}`];
  if (category) tags.push(`category/${category}`);
  if (source) tags.push(`source/${source.toLowerCase()}`);

  const created = tp.date.now("YYYY-MM-DD");
  const fm = [
    '---',
    `type: "вишлист"`,
    `title: "${title}"`,
    `created: ${created}`,
    'tags:',
    ...tags.map(t => `  - ${t}`),
    'attachments:',
    '  links:',
    link ? `  - "${link}"` : '  -',
    '  files:',
    'form: "link"',
    `priority: "${priority}"`,
    `price: "${price}"`,
    `reason: "${reason}"`,
    'raw_dir: ""',
    '---'
  ].join('\n');

  const body = [
    `# ${title}`,
    '',
    '- Исходные файлы: ',
    '',
    '## Почему хочу',
    reason ? `- ${reason}` : '- '
  ].join('\n');

  const content = `${fm}\n\n${body}\n`;
  const newFile = await tp.app.vault.create(filePath, content);
  const currentFile = tp.file.find_tfile(tp.file.title); if (currentFile) await tp.app.vault.delete(currentFile);
  await tp.app.workspace.getLeaf().openFile(newFile);
  new Notice(`✅ Вишлист: "${title}" добавлен!`);
} catch (error) {
  new Notice(`❌ Ошибка: ${error.message}`);
}
%>
