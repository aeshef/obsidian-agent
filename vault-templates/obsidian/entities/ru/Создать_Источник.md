<%*
try {
  const title = await tp.system.prompt("Название источника (канал/персона)");
  if (!title) { await tp.app.vault.delete(tp.file.find_tfile(tp.file.title)); return; }

  const platform = await tp.system.prompt("Платформа (telegram/youtube/...) (опц.)", "");
  const handle = await tp.system.prompt("Аккаунт/ссылка (опц.)", "");
  const topicsRaw = await tp.system.prompt("Темы (по одной в строке, topic/...) (опц.)", "");
  const quality = await tp.system.prompt("Качество 1–5 (опц.)", "");

  const topics = (topicsRaw || '').split(/\n+/).map(s => s.trim()).filter(Boolean);

  const slug = (s) => s.replace(/[^\p{L}\p{N}\s-_]/gu, "").trim().replace(/\s+/g, "_");
  const baseDir = `700_База_Данных/Источники`;
  const baseName = slug(title);
  let filePath = `${baseDir}/${baseName}.md`;
  while (tp.app.vault.getAbstractFileByPath(filePath)) { filePath = `${baseDir}/${baseName}-i.md`; }

  const tags = topics.map(t => `topic/${t}`);

  const created = tp.date.now("YYYY-MM-DD");
  const fm = [
    '---',
    `type: "источник"`,
    `title: "${title}"`,
    `created: ${created}`,
    'tags:',
    ...tags.map(t => `  - ${t}`),
    'attachments:',
    '  links:',
    handle ? `  - "${handle}"` : '  -',
    '  files:',
    `platform: "${platform}"`,
    `handle: "${handle}"`,
    'topics:',
    ...topics.map(t => `  - ${t}`),
    `quality: ${quality || ''}`,
    'raw_dir: ""',
    '---'
  ].join('\n');

  const body = [
    `# ${title}`,
    '',
    `- Платформа: ${platform}`,
    `- Аккаунт/ссылка: ${handle}`,
    '- Исходные файлы: ',
  ].join('\n');

  const content = `${fm}\n\n${body}\n`;
  const newFile = await tp.app.vault.create(filePath, content);
  const currentFile = tp.file.find_tfile(tp.file.title); if (currentFile) await tp.app.vault.delete(currentFile);
  await tp.app.workspace.getLeaf().openFile(newFile);
  new Notice(`✅ Источник "${title}" добавлен!`);
} catch (error) { new Notice(`❌ Ошибка: ${error.message}`); }
%>
