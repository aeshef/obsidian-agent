<%*
try {
  const title = await tp.system.prompt("Название курса");
  if (!title) { await tp.app.vault.delete(tp.file.find_tfile(tp.file.title)); return; }

  const platform = await tp.system.prompt("Платформа (Coursera/Udemy/...) (опц.)", "");
  const instructor = await tp.system.prompt("Преподаватель (опц.)", "");
  const status = await tp.system.suggester(["planned", "in-progress", "done"],["planned", "in-progress", "done"]) || "planned";
  const link = await tp.system.prompt("Ссылка на курс", "");

  const slug = (s) => s.replace(/[^\p{L}\p{N}\s-_]/gu, "").trim().replace(/\s+/g, "_");
  const baseDir = `700_База_Данных/Курсы`;
  const baseName = slug(title);
  let filePath = `${baseDir}/${baseName}.md`;
  while (tp.app.vault.getAbstractFileByPath(filePath)) { filePath = `${baseDir}/${baseName}-i.md`; }

  const tags = ['domain/study', `status/${status}`];

  const created = tp.date.now("YYYY-MM-DD");
  const fm = [
    '---',
    `type: "курс"`,
    `title: "${title}"`,
    `created: ${created}`,
    'tags:',
    ...tags.map(t => `  - ${t}`),
    'attachments:',
    '  links:',
    link ? `  - "${link}"` : '  -',
    '  files:',
    'form: "link"',
    `status: "${status}"`,
    `platform: "${platform}"`,
    `instructor: "${instructor}"`,
    'raw_dir: ""',
    '---'
  ].join('\n');

  const body = [
    `# ${title}`,
    '',
    '- Исходные файлы: ',
    '',
    '## Содержание/модульность',
    '- '
  ].join('\n');

  const content = `${fm}\n\n${body}\n`;
  const newFile = await tp.app.vault.create(filePath, content);
  const currentFile = tp.file.find_tfile(tp.file.title); if (currentFile) await tp.app.vault.delete(currentFile);
  await tp.app.workspace.getLeaf().openFile(newFile);
  new Notice(`✅ Курс "${title}" добавлен!`);
} catch (error) {
  new Notice(`❌ Ошибка: ${error.message}`);
}
%>
