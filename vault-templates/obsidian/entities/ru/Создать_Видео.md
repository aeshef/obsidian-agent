<%*
try {
  const title = await tp.system.prompt("Название видео/ролика");
  if (!title) { await tp.app.vault.delete(tp.file.find_tfile(tp.file.title)); return; }

  const author = await tp.system.prompt("Автор/канал", "");
  const status = await tp.system.suggester(["planned", "in-progress", "done"],["planned", "in-progress", "done"]) || "planned";
  const source = await tp.system.suggester(["youtube", "reels", "tiktok", "другое"],["youtube", "reels", "tiktok", "другое"]) || "другое";
  const link = await tp.system.prompt("Ссылка (YouTube/и т.п.)", "");
  const topic = await tp.system.prompt("Тема (topic/...) — одно слово", "");

  const slug = (s) => s
    .replace(/[^\p{L}\p{N}\s-_]/gu, "")
    .trim()
    .replace(/\s+/g, "_");

  const baseDir = `700_База_Данных/Видео`;
  const baseName = slug(title);
  let filePath = `${baseDir}/${baseName}.md`;
  while (tp.app.vault.getAbstractFileByPath(filePath)) { filePath = `${baseDir}/${baseName}-i.md`; }

  const tags = [];
  if (topic) tags.push(`topic/${topic.toLowerCase()}`);
  tags.push(`status/${status}`);
  tags.push(`source/${source.toLowerCase().replace(/\s+/g, '_')}`);

  const created = tp.date.now("YYYY-MM-DD");
  const fm = [
    '---',
    `type: "видео"`,
    `title: "${title}"`,
    `created: ${created}`,
    'tags:',
    ...tags.map(t => `  - ${t}`),
    'attachments:',
    '  links:',
    link ? `  - "${link}"` : '  -',
    '  files:',
    'source: "' + source + '"',
    'form: "link"',
    `status: "${status}"`,
    `author: "${author}"`,
    'rating: ',
    'raw_dir: ""',
    '---'
  ].join('\n');

  const body = [
    `# ${title}`,
    '',
    '- Исходные файлы: ',
    '',
    '## Ключевые тезисы',
    '- ',
    '',
    '## Инсайты',
    '- '
  ].join('\n');

  const content = `${fm}\n\n${body}\n`;
  const newFile = await tp.app.vault.create(filePath, content);
  const currentFile = tp.file.find_tfile(tp.file.title); if (currentFile) await tp.app.vault.delete(currentFile);
  await tp.app.workspace.getLeaf().openFile(newFile);
  new Notice(`✅ Видео "${title}" добавлено!`);
} catch (error) {
  new Notice(`❌ Ошибка: ${error.message}`);
}
%>
